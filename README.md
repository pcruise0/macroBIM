### REBAR INPUT

```javascript
{
  id: "SMART_REBAR_01", 
  code: 44,        // 철근 shape code (BS8666 기준)

  // 1. 초기 위치 및 회전 (init)
  // x, y 좌표를 직접 입력하여 허공에 생성할 경우, 
  init: { 
    x: "-WEB_X",     // 철근 생성 시 x 좌표 (변수 및 수식 적용 가능, 생략 가능)
    y: "BOTTOM_Y",   // 철근 생성 시 y 좌표 (생략 가능)
    rot: 10          // 철근 생성 시 회전 각도 
  },

  // 2. 세그먼트 상세 정의 (segs) - 길이(len)와 부착 벽체(set)를 한곳에서 직관적으로 관리!
  // ⭐ [강력 추천] 대소문자 구분 없이(a, b, c...) 자유롭게 입력 가능합니다.
  segs: {
    fix: "c",                  // 닻(Anchor) 세그먼트 지정: 'C' 구간을 절대 기준으로 삼아 부착! set이 2개 이상일때 필요, set 1개시 불필요
    a: { len: 500 },           // A 구간 길이
    b: { len: 1800 },          // B 구간 길이
    c: { len: 400, set: "E5" },// C 구간 길이 및 타겟 벽체 지정 (C 구간을 E5 벽에 오차 없이 밀착)
    d: { len: 1800 },          // D 구간 길이
    e: { len: 500 }            // E 구간 길이
  },

  // 3. 각도 및 중력 방향 (angs, nors) - 생략 시 자동 디폴트값 적용
  angs: { ra: 90, rb: 90, rc: 90, rd: 90 },  // 각 구간 사이의 꺾임 각도 (대소문자 무관)
  nors: { a: 1, b: -1, c: -1, d: -1, e: 1 }, // 물리 엔진 중력 방향 (set 부착 시 밖을 향하면 자동 반전됨!)

  // 4. 단부 처리 (ends - Begin / End)
  // fit: 현재 부착된 벽체의 피복선 끝단까지 정확히 연장
  // ray: 철근 축 방향으로 레이저를 쏴서 만나는 '다음 벽체'까지 연장 (+ 값은 추가 연장, - 값은 후퇴)
  ends: { 
    b: { "fit": 0 },   // 시작점(Begin). 대소문자 무관.
    e: { "ray": -40 }  // 끝점(End). 대소문자 무관.
  }
}
